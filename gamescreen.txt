import { useState, useCallback, useMemo } from 'react';
import { motion, AnimatePresence } from 'motion/react';
import { 
  Crown, 
  RotateCcw, 
  Pause, 
  History,
  X,
  Play
} from 'lucide-react';

// --- Types ---

type CellState = 'empty' | 'queen' | 'dot';

interface Position {
  row: number;
  col: number;
}

interface Level {
  id: string;
  gridSize: number;
  requiredQueens: number;
  regions?: number[][]; // Not used for this version but good for future
}

// --- Constants ---

const GRID_SIZE = 8;
const REQUIRED_QUEENS = 5;

// --- Components ---

export default function App() {
  const [board, setBoard] = useState<CellState[][]>(
    Array(GRID_SIZE).fill(null).map(() => Array(GRID_SIZE).fill('empty'))
  );
  const [isPaused, setIsPaused] = useState(false);

  // Initial setup to match screenshot partially
  useState(() => {
    const initialBoard = [...board];
    // Based on image roughly
    // initialBoard[0][3] = 'queen';
    // initialBoard[3][6] = 'queen';
    // initialBoard[5][1] = 'queen';
    // setBoard(initialBoard);
  });

  const queenCount = useMemo(() => {
    return board.flat().filter(cell => cell === 'queen').length;
  }, [board]);

  const toggleCell = useCallback((row: number, col: number) => {
    if (isPaused) return;
    
    setBoard(prev => {
      const next = prev.map(r => [...r]);
      const current = next[row][col];
      
      if (current === 'empty') next[row][col] = 'queen';
      else if (current === 'queen') next[row][col] = 'dot';
      else next[row][col] = 'empty';
      
      return next;
    });
  }, [isPaused]);

  const hasConflicts = (row: number, col: number) => {
    if (board[row][col] !== 'queen') return false;

    for (let i = 0; i < GRID_SIZE; i++) {
      // Row conflict
      if (i !== col && board[row][i] === 'queen') return true;
      // Col conflict
      if (i !== row && board[i][col] === 'queen') return true;
    }

    // Diagonal conflicts
    for (let i = 1; i < GRID_SIZE; i++) {
      if (row + i < GRID_SIZE && col + i < GRID_SIZE && board[row + i][col + i] === 'queen') return true;
      if (row - i >= 0 && col - i >= 0 && board[row - i][col - i] === 'queen') return true;
      if (row + i < GRID_SIZE && col - i >= 0 && board[row + i][col - i] === 'queen') return true;
      if (row - i >= 0 && col + i < GRID_SIZE && board[row - i][col + i] === 'queen') return true;
    }

    return false;
  };

  const resetGame = () => {
    setBoard(Array(GRID_SIZE).fill(null).map(() => Array(GRID_SIZE).fill('empty')));
  };

  return (
    <div className="flex flex-col items-center justify-center min-h-screen p-6 max-w-md mx-auto">
      {/* Header */}
      <motion.header 
        initial={{ opacity: 0, y: -20 }}
        animate={{ opacity: 1, y: 0 }}
        className="text-center mb-10 w-full"
      >
        <h1 className="font-display text-4xl font-bold tracking-[0.2em] text-primary mb-6">
          GAMEPLAY
        </h1>
        
        <div className="flex items-center justify-between w-full px-2">
          <div className="flex flex-col items-start gap-1">
            <span className="text-[10px] uppercase tracking-widest text-on-surface/50 font-semibold font-sans">
              STATUS
            </span>
            <div className="flex items-center gap-2">
              <div className="w-2 h-2 rounded-full bg-[#4682B4] shadow-[0_0_8px_rgba(70,130,180,0.6)]" />
              <span className="font-display font-semibold tracking-wide text-lg">
                YOUR TURN
              </span>
            </div>
          </div>

          <div className="glass px-4 py-3 rounded-2xl flex items-center gap-2">
            <Crown className="w-5 h-5 text-secondary" />
            <span className="font-sans font-semibold text-on-surface/80">
              Queens: {queenCount}/{REQUIRED_QUEENS}
            </span>
          </div>
        </div>
      </motion.header>

      {/* Main Board Card */}
      <motion.div 
        initial={{ opacity: 0, scale: 0.95 }}
        animate={{ opacity: 1, scale: 1 }}
        transition={{ delay: 0.1 }}
        className="glass rounded-[32px] w-full aspect-square relative overflow-hidden"
      >
        <div className="bg-white/80 w-full h-full grid grid-cols-8 grid-rows-8 border border-white/20">
          {board.map((row, rIdx) => (
            row.map((cell, cIdx) => (
              <Cell 
                key={`${rIdx}-${cIdx}`}
                state={cell}
                hasConflict={hasConflicts(rIdx, cIdx)}
                isLight={(rIdx + cIdx) % 2 === 0}
                onClick={() => toggleCell(rIdx, cIdx)}
              />
            ))
          ))}
        </div>

        {/* Pause Overlay */}
        <AnimatePresence>
          {isPaused && (
            <motion.div 
              initial={{ opacity: 0 }}
              animate={{ opacity: 1 }}
              exit={{ opacity: 0 }}
              className="absolute inset-0 z-10 bg-surface/40 backdrop-blur-md rounded-[32px] flex items-center justify-center"
            >
              <button 
                onClick={() => setIsPaused(false)}
                className="bg-primary text-white p-6 rounded-full shadow-2xl hover:scale-110 active:scale-95 transition-transform"
              >
                <Play className="w-10 h-10 fill-current" />
              </button>
            </motion.div>
          )}
        </AnimatePresence>
      </motion.div>

      {/* Footer Controls */}
      <footer className="mt-12 flex items-center justify-between w-full px-4">
        <motion.button 
          whileTap={{ scale: 0.9 }}
          onClick={() => setIsPaused(!isPaused)}
          className="w-14 h-14 rounded-full glass flex items-center justify-center text-primary/70 hover:text-primary transition-colors"
        >
          {isPaused ? <Play className="w-6 h-6" /> : <Pause className="w-6 h-6" />}
        </motion.button>

        <motion.button 
          whileHover={{ scale: 1.02 }}
          whileTap={{ scale: 0.95 }}
          onClick={resetGame}
          className="bg-primary text-white px-10 py-4 rounded-full font-display font-semibold tracking-wider shadow-xl shadow-black/10 active:bg-neutral-800 border-t border-white/20"
        >
          RESIGN
        </motion.button>

        <motion.button 
          whileTap={{ scale: 0.9 }}
          className="w-14 h-14 rounded-full glass flex items-center justify-center text-primary/70 hover:text-primary transition-colors"
        >
          <History className="w-6 h-6" />
        </motion.button>
      </footer>
    </div>
  );
}

// --- Subcomponents ---

interface CellProps {
  key?: string;
  state: CellState;
  hasConflict: boolean;
  isLight: boolean;
  onClick: () => void;
}

function Cell({ state, hasConflict, isLight, onClick }: CellProps) {
  return (
    <button 
      onClick={onClick}
      className={`
        w-full h-full relative flex items-center justify-center transition-colors duration-200
        ${isLight ? 'bg-white/40' : 'bg-[#edeeef]/40'}
        hover:bg-secondary/10
        ${hasConflict ? 'bg-error/10' : ''}
      `}
    >
      <AnimatePresence mode="wait">
        {state === 'queen' && (
          <motion.div
            key="queen"
            initial={{ scale: 0, rotate: -45 }}
            animate={{ scale: 1, rotate: 0 }}
            exit={{ scale: 0 }}
            transition={{ type: 'spring', damping: 15 }}
            className={`
              relative z-[2]
              ${hasConflict ? 'text-error' : 'text-primary'}
            `}
          >
            <Crown strokeWidth={2.5} className="w-3/5 h-3/5 mx-auto" />
          </motion.div>
        )}
        
        {state === 'dot' && (
          <motion.div
            key="dot"
            initial={{ scale: 0 }}
            animate={{ scale: 1 }}
            exit={{ scale: 0 }}
            className="w-2.5 h-2.5 bg-secondary/40 rounded-full border border-secondary/20 shadow-sm"
          />
        )}
      </AnimatePresence>
      
      {/* Visual background for dots/regions if needed - matching screenshot subtle glow */}
      {state === 'queen' && !hasConflict && (
        <div className="absolute inset-2 bg-secondary/5 rounded-md blur-sm" />
      )}
    </button>
  );
}
